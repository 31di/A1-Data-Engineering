    WITH

    LC_CTYPE = 'English_United States.1256'

    IS_TEMPLATE = False;




    -- Customers

-- Products
CREATE TABLE DimProduct (

-- Dates
CREATE TABLE DimDate (
    date_key NUMERIC PRIMARY KEY,
    full_date DATE,
    day INT,
    month INT,
    year INT,
    quarter INT,
    week INT
);

INSERT INTO public.dimdate (date_key, full_date, day, month, year, quarter, week)
SELECT
    (EXTRACT(YEAR FROM d)::INTEGER * 10000) +
    (EXTRACT(MONTH FROM d)::INTEGER * 100) +
    (EXTRACT(DAY FROM d)::INTEGER) AS date_key,
    d AS full_date,
    EXTRACT(DAY FROM d)::INTEGER AS day,
    EXTRACT(MONTH FROM d)::INTEGER AS month,
    EXTRACT(YEAR FROM d)::INTEGER AS year,
    EXTRACT(QUARTER FROM d)::INTEGER AS quarter,
    EXTRACT(WEEK FROM d)::INTEGER AS week
FROM generate_series('2015-01-01'::date, '2030-12-31'::date, '1 day') d;

CREATE TABLE public.dimtime (
    time_key        INTEGER PRIMARY KEY,
    full_time       TIME NOT NULL,
    hour            INTEGER NOT NULL,
    minute          INTEGER NOT NULL,
    second          INTEGER NOT NULL
);

INSERT INTO public.dimtime (time_key, full_time, hour, minute, second)
SELECT
    (EXTRACT(HOUR FROM t)::INTEGER * 10000) +
    (EXTRACT(MINUTE FROM t)::INTEGER * 100) +
    (EXTRACT(SECOND FROM t)::INTEGER) AS time_key,
    t::time AS full_time,
    EXTRACT(HOUR FROM t)::INTEGER AS hour,
    EXTRACT(MINUTE FROM t)::INTEGER AS minute,
    EXTRACT(SECOND FROM t)::INTEGER AS second
FROM generate_series(
        '2024-01-01 00:00:00'::timestamp,
        '2024-01-01 23:59:59'::timestamp,
        '1 second'::interval
    ) AS t;

-- FactOrder (Aggregated على مستوى الطلب)

    date_delivered_carrier_key INT,
    date_delivered_customer_key INT,

    time_delivered_customer_key INT,

-- FactSales
